  <script>
    // State Data Storage
    let db = {
      assets: ["EUR/USD", "GBP/USD", "USD/JPY", "AUD/CAD", "EUR/JPY", "GBP/JPY", "EUR/USD OTC", "GBP/USD OTC", "USD/INR OTC", "USD/BDT OTC", "USD/BRL OTC", "EUR/GBP OTC", "AUD/CHF OTC"],
      backtests: {}, // key: "asset_level_testIndex", value: "Continuation Candle" or "Reversal Candle"
      trades: [],    // array of live trade objects
      winStreak: 0,
      lossStreak: 0,
      totalLossesToday: 0,
      lockoutUntil: 0, // timestamp
      lockoutType: null // '15min', 'session', 'day'
    };

    // Load from localStorage on startup
    function loadStorage() {
      const saved = localStorage.getItem('quotex_gann_suite_db_v3');
      if (saved) {
        try {
          db = JSON.parse(saved);
        } catch(e) { console.error("Error loading storage", e); }
      }
    }

    function saveStorage() {
      localStorage.setItem('quotex_gann_suite_db_v3', JSON.stringify(db));
      updateAllUI();
    }

    // Tab Switching
    function switchTab(tab) {
      const backtestView = document.getElementById('tabBacktestView');
      const liveView = document.getElementById('tabLiveView');
      const navBacktest = document.getElementById('navBacktest');
      const navLive = document.getElementById('navLive');

      if (tab === 'backtest') {
        backtestView.classList.remove('hidden');
        liveView.classList.add('hidden');
        navBacktest.className = "px-3 py-1 rounded-md font-semibold transition-all bg-red-600 text-white shadow-sm";
        navLive.className = "px-3 py-1 rounded-md font-semibold transition-all text-gray-400 hover:text-white";
      } else {
        backtestView.classList.add('hidden');
        liveView.classList.remove('hidden');
        navLive.className = "px-3 py-1 rounded-md font-semibold transition-all bg-red-600 text-white shadow-sm";
        navBacktest.className = "px-3 py-1 rounded-md font-semibold transition-all text-gray-400 hover:text-white";
      }
    }

    // IST Clock & Session Detector (+5:30)
    function getISTTime() {
      const now = new Date();
      const utc = now.getTime() + (now.getTimezoneOffset() * 60000);
      const istTime = new Date(utc + (3600000 * 5.5));
      return istTime;
    }

    function updateClockAndSession() {
      const ist = getISTTime();
      const hours = ist.getHours();
      const minutes = ist.getMinutes();
      const seconds = ist.getSeconds();
      
      const timeStr = [
        hours.toString().padStart(2, '0'),
        minutes.toString().padStart(2, '0'),
        seconds.toString().padStart(2, '0')
      ].join(':');

      document.getElementById('istClock').innerText = timeStr;

      // Determine Market Session based on IST hours
      let sessionName = "Sydney / Tokyo";
      let badgeColor = "bg-purple-950 text-purple-400 border-purple-800/50";
      
      if (hours >= 12 && hours < 17) {
        sessionName = "London Open";
        badgeColor = "bg-amber-950 text-amber-400 border-amber-800/50";
      } else if (hours >= 17 && hours < 22) {
        sessionName = "London / NY Overlap";
        badgeColor = "bg-emerald-950 text-emerald-400 border-emerald-800/50";
      } else if (hours >= 22 || hours < 2) {
        sessionName = "New York Late";
        badgeColor = "bg-blue-950 text-blue-400 border-blue-800/50";
      }

      const badge = document.getElementById('badgeSession');
      badge.innerText = sessionName;
      badge.className = `px-1.5 py-0.5 rounded text-[10px] border font-bold ${badgeColor}`;

      const liveSessionDisplay = document.getElementById('liveSessionDisplay');
      if (liveSessionDisplay) liveSessionDisplay.value = sessionName;

      // Check Lockout timers
      checkLockoutStatus();
    }

    // Populate Asset Dropdowns
    function populateAssets() {
      const backtestSelect = document.getElementById('backtestAsset');
      const liveSelect = document.getElementById('liveAsset');
      
      const currentBacktestVal = backtestSelect.value;
      const currentLiveVal = liveSelect.value;

      let optionsHTML = '';
      db.assets.forEach(asset => {
        optionsHTML += `<option value="${asset}">${asset}</option>`;
      });

      backtestSelect.innerHTML = optionsHTML;
      liveSelect.innerHTML = optionsHTML;

      if (db.assets.includes(currentBacktestVal)) backtestSelect.value = currentBacktestVal;
      if (db.assets.includes(currentLiveVal)) liveSelect.value = currentLiveVal;
    }

    function addCustomAsset() {
      const input = document.getElementById('customAssetInput');
      const val = input.value.trim().toUpperCase();
      if (val && !db.assets.includes(val)) {
        db.assets.push(val);
        input.value = '';
        populateAssets();
        document.getElementById('backtestAsset').value = val;
        renderBacktestGrid();
        saveStorage();
      }
    }

    // Render Back-Testing Grid (Rows: 25%, 50%, 75% | Cols: Test #1 to #7)
    function renderBacktestGrid() {
      const asset = document.getElementById('backtestAsset').value;
      const tbody = document.getElementById('gannTableBody');
      const levels = ["25%", "50%", "75%"];
      let html = '';

      levels.forEach(level => {
        html += `<tr class="hover:bg-gray-900/40">`;
        html += `<td class="p-2.5 font-bold text-gray-300 text-left pl-3 border-r border-gray-800 bg-gray-900/30 flex items-center justify-between">
                   <span>${level} Gann Level</span>
                   <span class="text-[10px] text-gray-500 font-normal">1M</span>
                 </td>`;

        for (let i = 1; i <= 7; i++) {
          const key = `${asset}_${level}_Test${i}`;
          const currentVal = db.backtests[key] || "";
          
          let selectColor = "text-gray-400 bg-gray-900 border-gray-800";
          if (currentVal === "Continuation Candle") selectColor = "text-emerald-400 bg-emerald-950/40 border-emerald-800/60 font-semibold";
          if (currentVal === "Reversal Candle") selectColor = "text-red-400 bg-red-950/40 border-red-800/60 font-semibold";

          html += `<td class="p-1.5 border-r border-gray-800/60 text-center">
                     <select onchange="updateBacktestCell('${asset}', '${level}', ${i}, this.value)" class="w-full px-1.5 py-1 rounded text-xs border ${selectColor} focus:ring-1 focus:ring-red-500">
                       <option value="">-- Select --</option>
                       <option value="Continuation Candle" ${currentVal === 'Continuation Candle' ? 'selected' : ''}>Continuation Candle</option>
                       <option value="Reversal Candle" ${currentVal === 'Reversal Candle' ? 'selected' : ''}>Reversal Candle</option>
                     </select>
                   </td>`;
        }
        html += `</tr>`;
      });

      tbody.innerHTML = html;
      calculateAnalytics();
    }

    function updateBacktestCell(asset, level, testNum, val) {
      const key = `${asset}_${level}_Test${testNum}`;
      if (val === "") {
        delete db.backtests[key];
      } else {
        db.backtests[key] = val;
      }
      saveStorage();
      renderBacktestGrid();
    }

    // Analytics & Accuracy Calculations
    function calculateAnalytics() {
      let totalTests = 0;
      let totalContinuations = 0;
      let count25 = { total: 0, reversals: 0 };
      let count50 = { total: 0, reversals: 0 };
      let count75 = { total: 0, reversals: 0 };

      Object.keys(db.backtests).forEach(key => {
        const val = db.backtests[key];
        if (val) {
          totalTests++;
          if (val === "Continuation Candle") totalContinuations++;
          
          if (key.includes("_25%_")) {
            count25.total++;
            if (val === "Reversal Candle") count25.reversals++;
          } else if (key.includes("_50%_")) {
            count50.total++;
            if (val === "Reversal Candle") count50.reversals++;
          } else if (key.includes("_75%_")) {
            count75.total++;
            if (val === "Reversal Candle") count75.reversals++;
          }
        }
      });

      const winRate = totalTests > 0 ? ((totalContinuations / totalTests) * 100).toFixed(1) : "0.0";
      const acc25 = count25.total > 0 ? ((count25.reversals / count25.total) * 100).toFixed(1) : "0.0";
      const acc50 = count50.total > 0 ? ((count50.reversals / count50.total) * 100).toFixed(1) : "0.0";
      const acc75 = count75.total > 0 ? ((count75.reversals / count75.total) * 100).toFixed(1) : "0.0";

      document.getElementById('statWinRate').innerText = `${winRate}%`;
      document.getElementById('statTotalTests').innerText = `${totalTests} tests`;
      document.getElementById('statAcc25').innerText = `${acc25}%`;
      document.getElementById('statAcc50').innerText = `${acc50}%`;
      document.getElementById('statAcc75').innerText = `${acc75}%`;

      document.getElementById('headerWinStreak').innerText = `W: ${db.winStreak}`;
      document.getElementById('headerLossStreak').innerText = `L: ${db.lossStreak}`;
      document.getElementById('liveWinStreakBadge').innerText = db.winStreak;
      document.getElementById('liveLossStreakBadge').innerText = db.lossStreak;

      renderLiveTradesTable();
      renderAIAdvisor();
    }

    // Live Trade Logging & Risk Rules
    function recordLiveTrade(outcome) {
      if (db.lockoutUntil && Date.now() < db.lockoutUntil) {
        alert("Live Trading is currently locked due to risk management rules. Please wait.");
        return;
      }

      const ist = getISTTime();
      const timeStr = ist.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });

      const trade = {
        id: Date.now(),
        time: timeStr,
        asset: document.getElementById('liveAsset').value,
        session: document.getElementById('liveSessionDisplay').value,
        trend: document.getElementById('liveTrend').value,
        gannLevel: document.getElementById('liveGannLevel').value,
        pattern: document.getElementById('livePattern').value,
        prediction: document.getElementById('livePrediction').value,
        outcome: outcome
      };

      db.trades.unshift(trade);

      // Streak & Risk Management logic
      if (outcome === 'WIN') {
        db.winStreak++;
        db.lossStreak = 0;
      } else if (outcome === 'LOSS') {
        db.lossStreak++;
        db.winStreak = 0;
        db.totalLossesToday++;

        if (db.lossStreak === 2) {
          // 2 consecutive losses: 15 min cooldown
          db.lockoutUntil = Date.now() + (15 * 60 * 1000);
          db.lockoutType = '15min';
        } else if (db.lossStreak >= 3) {
          if (db.totalLossesToday >= 6) {
            // 3 consecutive losses second time in a day -> lock for the rest of the day
            const tomorrow = new Date(ist);
            tomorrow.setDate(tomorrow.getDate() + 1);
            tomorrow.setHours(0,0,0,0);
            db.lockoutUntil = tomorrow.getTime();
            db.lockoutType = 'day';
          } else {
            // 3 consecutive losses first time -> lock until next session
            db.lockoutUntil = Date.now() + (2 * 3600 * 1000);
            db.lockoutType = 'session';
          }
        }
      }

      saveStorage();
    }

    function checkLockoutStatus() {
      const banner = document.getElementById('riskBanner');
      const bannerMsg = document.getElementById('riskBannerMsg');
      const lockoutNotice = document.getElementById('liveLockoutNotice');

      if (db.lockoutUntil && Date.now() < db.lockoutUntil) {
        const remainingSec = Math.ceil((db.lockoutUntil - Date.now()) / 1000);
        const mins = Math.floor(remainingSec / 60);
        const secs = remainingSec % 60;

        banner.classList.remove('hidden');
        if (db.lockoutType === '15min') {
          bannerMsg.innerText = `⚠️ 15-MINUTE COOLDOWN ACTIVE (${mins}m ${secs}s remaining)`;
        } else if (db.lockoutType === 'session') {
          bannerMsg.innerText = `🔒 TRADING LOCKED UNTIL NEXT SESSION (${mins}m remaining)`;
        } else {
          bannerMsg.innerText = `🛑 DAILY TRADING LOCKOUT ACTIVE (Locked for the rest of the day)`;
          lockoutNotice.classList.remove('hidden');
        }
      } else {
        banner.classList.add('hidden');
        lockoutNotice.classList.add('hidden');
        if (db.lockoutUntil) {
          db.lockoutUntil = 0;
          db.lockoutType = null;
          saveStorage();
        }
      }
    }

    // AI Advisor Setup Suggestions based on Past Trades
    function renderAIAdvisor() {
      const advisorDiv = document.getElementById('aiAdvisorContent');
      if (db.trades.length < 3) {
        advisorDiv.innerHTML = `<span class="text-gray-400">💡 Log at least 3 live trades with outcomes (Win/Loss) to activate AI pattern recognition and setup suggestions.</span>`;
        return;
      }

      let wins = db.trades.filter(t => t.outcome === 'WIN');
      let winRate = ((wins.length / db.trades.length) * 100).toFixed(1);

      let levelCounts = {};
      wins.forEach(w => { levelCounts[w.gannLevel] = (levelCounts[w.gannLevel] || 0) + 1; });
      let bestLevel = Object.keys(levelCounts).reduce((a, b) => levelCounts[a] > levelCounts[b] ? a : b, '50%');

      advisorDiv.innerHTML = `
        <div class="grid grid-cols-2 gap-3">
          <div class="bg-emerald-950/30 p-2 rounded border border-emerald-800/40">
            <span class="text-emerald-400 font-bold block mb-1"><i class="fa-solid fa-circle-check mr-1"></i> Highest Win-Rate Setup:</span>
            <span>Focus on <strong>${bestLevel} Gann Level</strong> setups during high-volume sessions. Current recorded win rate: <strong class="text-emerald-400">${winRate}%</strong>.</span>
          </div>
          <div class="bg-red-950/30 p-2 rounded border border-red-800/40">
            <span class="text-red-400 font-bold block mb-1"><i class="fa-solid fa-triangle-exclamation mr-1"></i> High-Risk Loss Warning:</span>
            <span>Avoid entering counter-trend continuation setups when consecutive losses reach 2. Stick strictly to 15-minute cooldown rules.</span>
          </div>
        </div>
      `;
    }

    function exportJSONData() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(db, null, 2));
      const downloadAnchor = document.createElement('a');
      downloadAnchor.setAttribute("href", dataStr);

      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      downloadAnchor.remove();
    }

    function importJSONData(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const imported = JSON.parse(e.target.result);
          if (imported && imported.backtests && imported.trades) {
            db = imported;
            saveStorage();
            populateAssets();
            renderBacktestGrid();
            alert("Backup successfully restored!");
          } else {
            alert("Invalid backup file structure.");
          }
        } catch(err) {
          alert("Error parsing JSON file.");
        }
      };
      reader.readAsText(file);
    }
